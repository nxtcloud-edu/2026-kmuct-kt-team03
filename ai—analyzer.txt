import json
import os
from openai import OpenAI


# --------------------------------------------------
# OpenAI API 설정
# --------------------------------------------------

client = OpenAI(
    api_key=os.getenv("OPENAI_API_KEY")
)


def make_fallback_result(notice):
    """
    AI 분석에 실패했을 때 반환할 기본 결과.
    API 오류가 발생해도 전체 프로그램이 멈추지 않게 한다.
    """

    return {
        "title": notice.get("title", ""),
        "url": notice.get("url", ""),
        "date": notice.get("date"),
        "content": notice.get("content", ""),
        "source": notice.get("source", ""),

        "category": "기타",
        "deadline": None,
        "target": "확인 필요",
        "summary": "AI 분석에 실패했습니다.",
        "action": "원문 확인 필요",
        "relevance": 0,
        "reason": "AI 분석 중 오류가 발생했습니다."
    }


def normalize_relevance(value):
    """
    relevance를 항상 0~100 사이 정수로 만든다.
    """

    try:
        value = int(value)

    except (TypeError, ValueError):
        return 0

    if value < 0:
        return 0

    if value > 100:
        return 100

    return value


def analyze_notice(notice: dict, user: dict) -> dict:
    """
    크롤러에서 받은 공지 하나를
    사용자 정보와 비교하여 OpenAI로 분석한다.
    """

    # 잘못된 데이터가 들어와도 프로그램이 죽지 않게 처리
    if not isinstance(notice, dict):
        notice = {}

    if not isinstance(user, dict):
        user = {}

    # --------------------------------------------------
    # 크롤러에서 받은 공지 정보
    # --------------------------------------------------

    title = notice.get("title", "")
    url = notice.get("url", "")
    date = notice.get("date")
    content = notice.get("content", "")
    source = notice.get("source", "")

    # --------------------------------------------------
    # 사용자 정보
    # --------------------------------------------------

    school = user.get("school", "")
    major = user.get("major", "")
    grade = user.get("grade", "")
    activities = user.get("activities", [])
    interests = user.get("interests", [])
    custom_interests = user.get("custom_interests", [])

    # --------------------------------------------------
    # OpenAI에게 보낼 프롬프트
    # --------------------------------------------------

    prompt = f"""
당신은 대학생을 위한 개인 맞춤형 공지 분석 AI입니다.

아래 사용자 정보와 공지사항을 비교하여
공지의 핵심 정보를 추출하고 사용자와의 관련성을 판단하세요.


[사용자 정보]

학교: {school}
학과: {major}
학년: {grade}
원하는 활동 유형: {activities}
관심 기술 분야: {interests}
기타 관심 분야: {custom_interests}


[공지사항]

제목: {title}
출처: {source}
작성일 또는 마감일: {date}

본문:
{content}


[분석 항목]

1. category

공지의 종류를 판단하세요.

다음 중 가장 적절한 종류를 선택하세요.

- 장학금
- 인턴
- 공모전
- 취업
- 교육
- 특강
- 교내활동
- 대외활동
- 과제
- 기타


2. deadline

신청, 접수 또는 과제 제출 마감일을 추출하세요.

가능하면 YYYY-MM-DD 형식으로 반환하세요.

마감일을 공지에서 확인할 수 없다면
null을 반환하세요.


3. target

공지에 실제로 명시되어 있는
지원 대상 또는 참가 대상을 간단하게 정리하세요.

지원 대상이 명시되어 있지 않다면
"명시되지 않음"이라고 작성하세요.


4. summary

공지의 핵심 내용을 사용자가 빠르게 이해할 수 있도록
1~2문장으로 요약하세요.


5. action

사용자가 이 공지에 관심이 있을 경우
해야 할 행동을 간단하게 작성하세요.

예:
"9월 25일까지 온라인 지원서 제출"

신청 방법이나 행동을 알 수 없다면
"원문 확인"이라고 작성하세요.


6. relevance

이 공지가 사용자에게 얼마나 관련 있는지
0~100 사이의 정수로 평가하세요.

다음을 종합적으로 고려하세요.

- 사용자 학교
- 사용자 학과
- 사용자 학년
- 사용자가 원하는 활동 유형
- 관심 기술 분야
- 기타 관심 분야
- 공지의 실제 지원 대상

사용자가 명확한 필수 지원 조건을 충족하지 못한다면
relevance를 낮게 평가하세요.

사용자가 실제 지원 대상이고
활동 유형과 관심 분야까지 잘 맞는다면
높은 relevance를 부여하세요.

단순히 같은 단어가 포함되어 있다는 이유만으로
높은 점수를 주지 마세요.

공지의 실제 의미를 고려하세요.


7. reason

왜 해당 relevance 점수를 주었는지
사용자 정보와 공지 내용을 비교하여
1~2문장으로 설명하세요.


[중요 규칙]

공지에 존재하지 않는 정보를 만들어내지 마세요.

지원 자격, 날짜, 혜택, 신청 방법 등을
임의로 추측하지 마세요.

확인할 수 없는 정보는
null 또는 "명시되지 않음"으로 처리하세요.

공지에 특정 학과나 학년 제한이 없다면
제한이 있다고 가정하지 마세요.


[출력 형식]

반드시 아래 구조의 JSON만 출력하세요.

추가 설명이나 마크다운 코드 블록은 작성하지 마세요.

{{
    "category": "문자열",
    "deadline": "YYYY-MM-DD 또는 null",
    "target": "문자열",
    "summary": "문자열",
    "action": "문자열",
    "relevance": 0,
    "reason": "문자열"
}}
"""

    try:

        # --------------------------------------------------
        # OpenAI에게 분석 요청
        # --------------------------------------------------

        response = client.responses.create(
            model="gpt-5.6-luna",
            input=prompt
        )

        response_text = response.output_text.strip()

        # 혹시 AI가 ```json 코드 블록을 붙인 경우 제거
        if response_text.startswith("```"):
            response_text = response_text.replace(
                "```json",
                "",
                1
            )

            response_text = response_text.replace(
                "```",
                ""
            )

            response_text = response_text.strip()

        # JSON 문자열을 Python dictionary로 변환
        ai_result = json.loads(response_text)

        category = ai_result.get(
            "category",
            "기타"
        )

        deadline = ai_result.get(
            "deadline"
        )

        target = ai_result.get(
            "target",
            "명시되지 않음"
        )

        summary = ai_result.get(
            "summary",
            ""
        )

        action = ai_result.get(
            "action",
            "원문 확인"
        )

        relevance = normalize_relevance(
            ai_result.get("relevance", 0)
        )

        reason = ai_result.get(
            "reason",
            "관련성 판단 이유를 확인할 수 없습니다."
        )

        # 크롤러 정보 + AI 분석 결과
        result = {
            "title": title,
            "url": url,
            "date": date,
            "content": content,
            "source": source,

            "category": category,
            "deadline": deadline,
            "target": target,
            "summary": summary,
            "action": action,
            "relevance": relevance,
            "reason": reason
        }

        return result

    except Exception as error:

        print(
            f"[AI 분석 실패] {title} - {error}"
        )

        return make_fallback_result(notice)
